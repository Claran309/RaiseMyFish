//go:build integration

package database

import (
 "testing"
 "time"
 "RaiseMyFish/internal/store"
)

func TestReviewSummarySourcePrivacyRevocation(t *testing.T){
 db,ctx:=repositoryDB(t)
 events,err:=NewEventRepo(db);if err!=nil{t.Fatal(err)}
 viewers,err:=NewViewerRepo(db);if err!=nil{t.Fatal(err)}
 base:=time.Now().UTC().Add(-2*time.Hour)
 seedSummaryViewer(t,events,"5005",base)
 opts:=store.ViewerRebuildOptions{MinHits:2,RecentTexts:15,RecentReplies:10}
 if _,err:=viewers.RebuildProfiles(ctx,opts);err!=nil{t.Fatal(err)}
 at:=base.Add(10*time.Minute)
 rec:=summaryRecord("5005","侧写观众","偏好：SYNTHETIC_REVOKED_SOURCE",at,[]string{"live:30308220:s:5005-1","live:30308220:s:5005-2"})
 if err:=viewers.SaveProfileSummary(ctx,"5005",rec.Summary,at,rec);err!=nil{t.Fatal(err)}
 // 增加其他公开资料保持档案仍达门槛；删除旧侧写引用的所有源事件。
 seedViewerEvent(t,events,"new:1","5005","侧写观众","in","普通公开一句","",base.Add(time.Minute*20))
 seedViewerEvent(t,events,"new:2","5005","侧写观众","in","普通公开二句","",base.Add(time.Minute*21))
 if err:=db.WithContext(ctx).Exec("DELETE FROM events WHERE id IN ?",rec.EventIDs).Error;err!=nil{t.Fatal(err)}
 if _,err:=viewers.RebuildProfiles(ctx,opts);err!=nil{t.Fatal(err)}
 p,err:=viewers.GetProfile(ctx,"5005");if err!=nil{t.Fatal(err)}
 var saved struct{ Summary string; Archived bool }
 if err:=db.Table("memory_records").Select("summary,archived").Where("id=?",rec.ID).Take(&saved).Error;err!=nil{t.Fatal(err)}
 t.Logf("all cited sources deleted: profile_summary_present=%v, memory_summary_present=%v, archived=%v",p.ProfileSummary!="",saved.Summary!="",saved.Archived)
 if p.ProfileSummary!="" || (saved.Summary!=""&&!saved.Archived){t.Fatal("derived public profile remains valid after all cited sources revoked")}
}
