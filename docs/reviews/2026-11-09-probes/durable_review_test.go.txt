func TestReview20261109DMEmptyEnvelopeMustNotSucceed(t *testing.T) {
 posts:=0
 srv:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){posts++; w.Header().Set("Content-Type","application/json"); _,_=w.Write([]byte(`{}`))}))
 defer srv.Close()
 client,err:=bili.New(bili.Options{APIRoot:srv.URL,LiveRoot:srv.URL,VCRoot:srv.URL,Cookie:"SESSDATA=fixture; bili_jct=fixture",SelfMID:"10001",Timeout:time.Second})
 if err!=nil {t.Fatal(err)}
 id,err:=client.SendMessage(context.Background(),"20002","offline probe")
 t.Logf("observed: HTTP200 {} -> msgID=%q err=%v POST=%d",id,err,posts)
 if err==nil {t.Fatal("BUG reproduced: empty envelope treated as successful DM response")}
}
